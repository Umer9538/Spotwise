rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // Allow authenticated users to read/write all collections
    // WARNING: Use only for development/testing!
    match /{document=**} {
      allow read, write: if request.auth != null;
    }
  }
}
